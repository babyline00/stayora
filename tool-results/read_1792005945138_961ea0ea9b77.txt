    50→export function HostDashboard() {
    51→  const { user, hostTab, setHostTab, setView } = useAppStore()
    52→  const [listings, setListings] = useState<any[]>([])
    53→  const [bookings, setBookings] = useState<any[]>([])
    54→  const [payouts, setPayouts] = useState<any[]>([])
    55→  const [stats, setStats] = useState<any>(null)
    56→  const [loading, setLoading] = useState(true)
    57→
    58→  const refreshAll = () => {
    59→    setLoading(true)
    60→    fetchHostDashboard()
    61→      .then((d: any) => {
    62→        setListings(d.listings || [])
    63→        setBookings(d.bookings || [])
    64→        setPayouts(d.payouts || [])
    65→        setStats(d.stats || null)
    66→      })
    67→      .catch(() => {
    68→        setListings([])
    69→        setBookings([])
    70→        setPayouts([])
    71→        setStats(null)
    72→      })
    73→      .finally(() => setLoading(false))
    74→  }
    75→
    76→  // Initial data fetch when the host user changes. Defers setState to a microtask
    77→  // so we never call setState synchronously in the effect body (avoids cascading renders).
    78→  useEffect(() => {
    79→    let cancelled = false
    80→    Promise.resolve()
    81→      .then(() => { if (!cancelled) refreshAll() })
    82→    return () => { cancelled = true }
    83→  }, [user?.id])
    84→
    85→  if (!user) return null
    86→
    87→  const handleNav = (tab: HostTab) => {
    88→    if (tab === 'create') {
    89→      // Create-listing wizard lives in the parent app — switch views.
    90→      setView('create-listing')
    91→      return
    92→    }
    93→    setHostTab(tab)
    94→  }
    95→
    96→  // Pending reservations count badge
    97→  const pendingReservations = bookings.filter((b: any) => b.status === 'PENDING').length
    98→
    99→  return (
   100→    <div className="mx-auto flex max-w-7xl gap-6 px-4 py-6 sm:px-6">
   101→      {/* Left sidebar */}
   102→      <aside className="hidden w-60 shrink-0 lg:block">
   103→        <div className="sticky top-20 rounded-2xl border border-border bg-white p-2">
   104→          <HostProfileCard user={user} />
   105→          <nav className="mt-2 space-y-0.5">
   106→            {TABS.map(t => {
   107→              const Icon = t.icon
   108→              const active = hostTab === t.id
   109→              return (
   110→                <button
   111→                  key={t.id}
   112→                  onClick={() => handleNav(t.id)}
   113→                  className={`flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition ${
   114→                    active ? 'bg-navy text-white' : 'text-navy/80 hover:bg-peach-light/60 hover:text-navy'
   115→                  }`}
   116→                >
   117→                  <Icon className="h-4 w-4" />
   118→                  <span className="flex-1 text-left">{t.label}</span>
   119→                  {t.id === 'reservations' && pendingReservations > 0 && (
   120→                    <Badge variant="secondary" className={`text-[10px] ${active ? 'bg-white/20 text-white' : 'bg-orange text-white'}`}>{pendingReservations}</Badge>
   121→                  )}
   122→                </button>
   123→              )
   124→            })}
   125→          </nav>
   126→        </div>
   127→      </aside>
   128→
   129→      {/* Main content */}
   130→      <main className="min-w-0 flex-1">
   131→        {/* Mobile tab pills */}
   132→        <div className="mb-4 flex gap-1 overflow-x-auto pb-1 lg:hidden stayora-scroll">
   133→          {TABS.map(t => {
   134→            const Icon = t.icon
   135→            const active = hostTab === t.id
   136→            return (
   137→              <button
   138→                key={t.id}
   139→                onClick={() => handleNav(t.id)}
   140→                className={`flex shrink-0 items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-medium ${
   141→                  active ? 'bg-navy text-white' : 'bg-peach-light/60 text-navy'
   142→                }`}
   143→              >
   144→                <Icon className="h-3.5 w-3.5" /> {t.label}
   145→              </button>
   146→            )
   147→          })}
   148→        </div>
   149→
   150→        {hostTab === 'overview' && (
   151→          <OverviewTab listings={listings} bookings={bookings} payouts={payouts} stats={stats} loading={loading} onGoToTab={setHostTab} onCreateListing={() => setView('create-listing')} />
   152→        )}
   153→        {hostTab === 'listings' && (
   154→          <ListingsTab listings={listings} loading={loading} onRefresh={refreshAll} onCreateListing={() => setView('create-listing')} />
   155→        )}
   156→        {hostTab === 'calendar' && <CalendarTab listings={listings} bookings={bookings} loading={loading} />}
   157→        {hostTab === 'reservations' && <ReservationsTab bookings={bookings} loading={loading} onRefresh={refreshAll} />}
   158→        {hostTab === 'inbox' && <InboxTab loading={loading} />}
   159→        {hostTab === 'earnings' && <EarningsTab bookings={bookings} payouts={payouts} stats={stats} loading={loading} />}
   160→        {hostTab === 'analytics' && <AnalyticsTab listings={listings} bookings={bookings} loading={loading} />}
   161→        {hostTab === 'reviews' && <ReviewsTab bookings={bookings} loading={loading} />}
   162→        {hostTab === 'cohosts' && <CoHostsTab listings={listings} />}
   163→        {hostTab === 'create' && <CreateListingPlaceholder onCreateListing={() => setView('create-listing')} />}
   164→      </main>
   165→    </div>
   166→  )
   167→}

   404→function ListingsTab({ listings, loading, onRefresh, onCreateListing }: any) {
   405→  const { showToast } = useAppStore()
   406→  const [filter, setFilter] = useState<'ALL' | 'DRAFT' | 'PUBLISHED' | 'PAUSED'>('ALL')
   407→  const [toggling, setToggling] = useState<string | null>(null)
   408→  const [duplicating, setDuplicating] = useState<string | null>(null)
   409→
   410→  const filtered = filter === 'ALL' ? listings : listings.filter((l: any) => l.status === filter)
   411→
   412→  const handleToggleStatus = async (listing: any) => {
   413→    const next = listing.status === 'PUBLISHED' ? 'PAUSED' : 'PUBLISHED'
   414→    setToggling(listing.id)
   415→    try {
   416→      await updateListingStatus(listing.id, next)
   417→      showToast(
   418→        next === 'PUBLISHED' ? 'Listing published' : 'Listing paused',
   419→        next === 'PUBLISHED' ? 'Your listing is now live and bookable.' : 'Bookings are paused for this listing.',
   420→        'success',
   421→      )
   422→      onRefresh()
   423→    } catch (e: any) {
   424→      showToast('Could not update status', e?.message, 'error')
   425→    } finally {
   426→      setToggling(null)
   427→    }
   428→  }
   429→
   430→  const handleDuplicate = async (listing: any) => {
   431→    setDuplicating(listing.id)
   432→    try {
   433→      await createListing({
   434→        title: `${listing.title} (copy)`,
   435→        description: listing.description,
   436→        category: listing.category,
   437→        roomType: listing.roomType,
   438→        country: listing.country,
   439→        city: listing.city,

   616→function CalendarTab({ listings, bookings, loading }: any) {
   617→  const { showToast } = useAppStore()
   618→  const [selectedListingId, setSelectedListingId] = useState<string>('')
   619→  const [selectedDate, setSelectedDate] = useState<Date | undefined>(new Date())
   620→  const [blocked, setBlocked] = useState<Set<string>>(new Set())
   621→
   622→  // Derive the effective listing without a setState-in-effect (avoids cascading renders).
   623→  const effectiveListingId = selectedListingId || (listings[0]?.id ?? '')
   624→
   625→  if (loading) return <Skeleton className="h-96 w-full rounded-2xl" />
   626→
   627→  if (listings.length === 0) {
   628→    return <EmptyState icon={CalendarIcon} title="No listings to schedule" subtitle="Create a listing to manage its availability calendar." />
   629→  }
   630→
   631→  const selectedListing = listings.find((l: any) => l.id === effectiveListingId) || listings[0]
   632→
   633→  // Compute reserved date set for the selected listing
   634→  const reservedSet = new Set<string>()
   635→  for (const b of bookings) {
   636→    if (b.listingId !== selectedListing.id) continue
   637→    if (b.status === 'CANCELLED') continue
   638→    const start = new Date(b.checkIn)
   639→    const end = new Date(b.checkOut)
   640→    const cur = new Date(start)
   641→    while (cur < end) {
   642→      reservedSet.add(cur.toISOString().split('T')[0])
   643→      cur.setDate(cur.getDate() + 1)
   644→    }
   645→  }
   646→  // Merge blocked (host-blocked) into reserved view
   647→  const allBlocked = new Set<string>([...reservedSet, ...blocked])
   648→
   649→  const dayStatus = (date: Date): 'available' | 'reserved' | 'blocked' => {
   650→    const key = date.toISOString().split('T')[0]
   651→    if (reservedSet.has(key)) return 'reserved'
   652→    if (blocked.has(key)) return 'blocked'
   653→    return 'available'
   654→  }
   655→
   656→  const toggleBlock = () => {
   657→    if (!selectedDate) {
   658→      showToast('Pick a date first', undefined, 'error')
   659→      return
   660→    }
   661→    const key = selectedDate.toISOString().split('T')[0]
   662→    const next = new Set(blocked)
   663→    if (next.has(key)) {
   664→      next.delete(key)
   665→      showToast('Date unblocked', 'Guests can now book this date.', 'success')
   666→    } else {
   667→      next.add(key)
   668→      showToast('Date blocked', 'Guests cannot book this date.', 'default')
   669→    }
   670→    setBlocked(next)
   671→  }
   672→
   673→  const selectedStatus = selectedDate ? dayStatus(selectedDate) : 'available'
   674→
   675→  return (
   676→    <div className="space-y-6">
   677→      <div>
   678→        <h1 className="text-2xl font-bold text-navy">Calendar</h1>
   679→        <p className="text-sm text-muted-foreground">Manage availability and block dates for individual listings.</p>
   680→      </div>
   681→
   682→      <div className="grid gap-4 lg:grid-cols-[1fr_320px]">
   683→        <Card>
   684→          <CardHeader>
   685→            <CardTitle className="flex items-center justify-between text-navy">
   686→              <span className="flex items-center gap-2"><CalendarIcon className="h-5 w-5 text-orange" /> Availability</span>
   687→              <Select value={effectiveListingId} onValueChange={setSelectedListingId}>
   688→                <SelectTrigger className="h-8 w-[220px] rounded-lg"><SelectValue placeholder="Select listing" /></SelectTrigger>
   689→                <SelectContent>
   690→                  {listings.map((l: any) => (
   691→                    <SelectItem key={l.id} value={l.id}>{l.title}</SelectItem>
   692→                  ))}
   693→                </SelectContent>
   694→              </Select>
   695→            </CardTitle>
   696→          </CardHeader>
   697→          <CardContent className="flex justify-center">
   698→            <Calendar
   699→              mode="single"
   700→              selected={selectedDate}
   701→              onSelect={setSelectedDate}
   702→              className="rounded-lg"
   703→              modifiers={{
   704→                reserved: (date: Date) => {
   705→                  const key = date.toISOString().split('T')[0]
   706→                  return reservedSet.has(key)
   707→                },
   708→                blocked: (date: Date) => {
   709→                  const key = date.toISOString().split('T')[0]
   710→                  return blocked.has(key)
   711→                },
   712→              }}
   713→              modifiersStyles={{
   714→                reserved: { backgroundColor: '#1a2c4f', color: '#ffffff', borderRadius: '6px' },
   715→                blocked: { backgroundColor: '#fce8d8', color: '#1a2c4f', textDecoration: 'line-through', borderRadius: '6px' },
   716→              }}
   717→            />
   718→          </CardContent>
   719→        </Card>
   720→
   721→        <div className="space-y-4">
   722→          <Card>
   723→            <CardHeader><CardTitle className="text-navy">Selected date</CardTitle></CardHeader>
   724→            <CardContent className="space-y-3 text-sm">
   725→              {selectedDate ? (
   726→                <>
   727→                  <div>
   728→                    <p className="text-xs text-muted-foreground">Date</p>

   730→                  </div>
   731→                  <div>
   732→                    <p className="text-xs text-muted-foreground">Status</p>

   739→                  <Button onClick={toggleBlock} className="w-full rounded-full bg-orange text-white">
   740→                    {selectedStatus === 'blocked' ? 'Unblock date' : 'Block date'}
   741→                  </Button>
   742→                  <p className="text-[10px] text-muted-foreground">Block dates to prevent new bookings (e.g., for personal use or maintenance).</p>

  1359→      setResponseBody('')

  1378→      </div>
  1379→
  1380→      <div className="flex flex-wrap gap-2">
  1381→        {(['all', '5', '4', '3'] as const).map(s => (

  1394→      {filtered.length === 0 ? (

  1396→      ) : (
  1397→        <div className="space-y-3">
  1398→          {filtered.map((r: any) => (
  1399→            <Card key={r.id}>
  1400→              <CardContent className="p-4">
  1401→                <div className="flex items-start gap-3">
  1402→                  <Avatar className="h-10 w-10">

  1407→                    <div className="flex flex-wrap items-center gap-2">
  1408→                      <p className="text-sm font-semibold text-navy">{r.author?.name}</p>
  1409→                      <div className="flex items-center gap-0.5">
  1410→                        {Array.from({ length: 5 }).map((_, i) => (
  1411→                          <Star key={i} className={`h-3 w-3 ${i < r.rating ? 'fill-orange text-orange' : 'text-muted-foreground/30'}`} />

  1450→            </Card>
  1451→          ))}

  1454→    </div>
