     6→interface SafeUser {
     7→  id: string
     8→  email: string
     9→  name: string
    10→  avatarUrl?: string | null
    11→  role: string
    12→  bio?: string | null
    13→  country?: string | null
    14→  isVerified: boolean
    15→  isSuperhost: boolean
    16→  responseRate: number
    17→  responseTimeMins: number
    18→}
    19→
    20→function toSafeUser(user: any): SafeUser {
    21→  return {
    22→    id: user.id,
    23→    email: user.email,
    24→    name: user.name,
    25→    avatarUrl: user.avatarUrl,
    26→    role: user.role,
    27→    bio: user.bio,
    28→    country: user.country,
    29→    isVerified: user.isVerified,
    30→    isSuperhost: user.isSuperhost,
    31→    responseRate: user.responseRate,
    32→    responseTimeMins: user.responseTimeMins,
    33→  }
    34→}
    35→
    36→async function getSessionUser(): Promise<SafeUser | null> {
    37→  const c = await cookies()
    38→  const token = c.get('stayora-session')?.value
    39→  if (!token) return null
    40→  const session = await db.session.findUnique({
    41→    where: { token },
    42→    include: { user: true },
    43→  })
    44→  if (!session || session.expiresAt < new Date()) return null
    45→  return toSafeUser(session.user)
    46→}
    47→
    48→async function requireAuth(roles?: string[]): Promise<SafeUser | null> {
    49→  const user = await getSessionUser()
    50→  if (!user) return null
    51→  if (roles && !roles.includes(user.role)) return null
    52→  return user
    53→}

   328→            upcomingCheckIns, occupancyRate,

   371→      case 'users': {
   372→        const me = await requireAuth(['ADMIN'])
   373→        if (!me) return NextResponse.json({ error: 'forbidden' }, { status: 403 })
   374→        const users = await db.user.findMany({

   381→        const code = url.searchParams.get('code') || ''
   382→        const subtotal = parseInt(url.searchParams.get('subtotal') || '0')

   385→        if (coupon.validTo < new Date()) return NextResponse.json({ valid: false, message: 'Coupon expired' })
   386→        if (coupon.minAmount > subtotal) return NextResponse.json({ valid: false, message: `Minimum spend is $${(coupon.minAmount / 100).toFixed(2)}` })
   387→        if (coupon.maxRedemptions && coupon.redeemed >= coupon.maxRedemptions) return NextResponse.json({ valid: false, message: 'Coupon fully redeemed' })
   388→        let discount = 0
   389→        if (coupon.type === 'PERCENT') discount = Math.round((subtotal * coupon.value) / 100)

   391→        return NextResponse.json({ valid: true, discount, coupon })
   392→      }
   393→      default:
   394→        return NextResponse.json({ error: 'unknown op' }, { status: 400 })
   395→    }
   396→  } catch (err: any) {
   397→    console.error('[stayora API GET error]', op, err)
   398→    return NextResponse.json({ error: err.message || 'server error' }, { status: 500 })
   399→  }
   400→}

   402→export async function POST(req: NextRequest) {
   403→  const url = new URL(req.url)
   404→  const op = url.searchParams.get('op') || ''
   405→  try {
   406→    const body = await req.json().catch(() => ({}))
   407→    switch (op) {
   408→      case 'register': {
   409→        const { email, password, name } = body
   410→        if (!email || !password || !name) return NextResponse.json({ error: 'Email, password, and name are required' }, { status: 400 })
   411→        if (password.length < 6) return NextResponse.json({ error: 'Password must be at least 6 characters' }, { status: 400 })
   412→        const existing = await db.user.findUnique({ where: { email } })
   413→        if (existing) return NextResponse.json({ error: 'Email already registered' }, { status: 409 })
   414→        const passwordHash = await bcrypt.hash(password, 10)
   415→        const role = (body.role || 'GUEST').toUpperCase()
   416→        const user = await db.user.create({
   417→          data: {
   418→            email, passwordHash, name,
   419→            role: ['GUEST', 'HOST'].includes(role) ? role : 'GUEST',
   420→            avatarUrl: body.avatarUrl || `https://i.pravatar.cc/200?u=${encodeURIComponent(email)}`,
   421→            bio: body.bio || null,
   422→            country: body.country || null,
   423→            isVerified: true,
   424→          },
   425→        })
   426→        const token = await createSession(user.id)
   427→        const c = await cookies()
   428→        c.set('stayora-session', token, { httpOnly: true, sameSite: 'lax', maxAge: 60 * 60 * 24 * 30, path: '/' })
   429→        return NextResponse.json({ user: toSafeUser(user) })
   430→      }
   431→      case 'login': {
   432→        const { email, password } = body
   433→        if (!email || !password) return NextResponse.json({ error: 'Email and password required' }, { status: 400 })
   434→        const user = await db.user.findUnique({ where: { email } })
   435→        if (!user) return NextResponse.json({ error: 'Invalid email or password' }, { status: 401 })
   436→        const ok = await bcrypt.compare(password, user.passwordHash)
   437→        if (!ok) return NextResponse.json({ error: 'Invalid email or password' }, { status: 401 })
   438→        const token = await createSession(user.id)
   439→        const c = await cookies()
   440→        c.set('stayora-session', token, { httpOnly: true, sameSite: 'lax', maxAge: 60 * 60 * 24 * 30, path: '/' })
   441→        return NextResponse.json({ user: toSafeUser(user) })
   442→      }
   443→      case 'logout': {
   444→        const me = await getSessionUser()
   445→        if (me) {
   446→          const c = await cookies()
   447→          c.delete('stayora-session')
   448→          await db.session.deleteMany({ where: { userId: me.id } }).catch(() => {})
   449→        }
   450→        return NextResponse.json({ ok: true })
   451→      }
   452→      case 'create-booking': {
   453→        const me = await requireAuth()
   454→        if (!me) return NextResponse.json({ error: 'Please sign in to book' }, { status: 401 })
   455→        const { listingId, checkIn, checkOut, adults, children, infants, pets, couponCode } = body
   456→        if (!listingId || !checkIn || !checkOut) return NextResponse.json({ error: 'Missing required fields' }, { status: 400 })
   457→        const listing = await db.listing.findUnique({ where: { id: listingId } })
   458→        if (!listing) return NextResponse.json({ error: 'Listing not found' }, { status: 404 })
   459→        const checkInDate = new Date(checkIn)
   460→        const checkOutDate = new Date(checkOut)
   461→        const nights = Math.ceil((checkOutDate.getTime() - checkInDate.getTime()) / 86400000)
   462→        if (nights < listing.minNights) return NextResponse.json({ error: `Minimum stay is ${listing.minNights} nights` }, { status: 400 })
   463→        if (listing.maxGuests < adults + children) return NextResponse.json({ error: 'Exceeds maximum occupancy' }, { status: 400 })
   464→        const conflicting = await db.availability.findFirst({
   465→          where: { listingId, date: { gte: checkInDate, lt: checkOutDate }, isAvailable: false },
   466→        })
   467→        if (conflicting) return NextResponse.json({ error: 'Selected dates are not available' }, { status: 409 })
   468→        const overlapping = await db.booking.findFirst({
   469→          where: { listingId, status: { in: ['CONFIRMED', 'PENDING'] }, checkIn: { lt: checkOutDate }, checkOut: { gt: checkInDate } },
   470→        })
   471→        if (overlapping) return NextResponse.json({ error: 'Selected dates overlap an existing booking' }, { status: 409 })
   472→        const nightlySubtotal = listing.basePrice * nights
   473→        const cleaningFee = listing.cleaningFee
   474→        const serviceFee = Math.round(nightlySubtotal * 0.12)
   475→        const taxes = Math.round(nightlySubtotal * 0.08)
   476→        let discounts = 0
   477→        if (couponCode) {
   478→          const coupon = await db.coupon.findUnique({ where: { code: couponCode } })
   479→          if (coupon && coupon.active && coupon.validTo > new Date() && coupon.minAmount <= nightlySubtotal) {
   480→            if (coupon.type === 'PERCENT') discounts = Math.round((nightlySubtotal * coupon.value) / 100)
   481→            else discounts = coupon.value
   482→          }
   483→        }
   484→        const totalAmount = nightlySubtotal + cleaningFee + serviceFee + taxes - discounts
   485→        const platformFee = serviceFee
   486→        const hostPayout = nightlySubtotal + cleaningFee - Math.round(nightlySubtotal * 0.03)
   487→        const reference = `STY-${Date.now().toString().slice(-7)}`
   488→        const booking = await db.booking.create({
   489→          data: {
   490→            reference, listingId, guestId: me.id, hostId: listing.hostId,
   491→            status: listing.instantBook ? 'CONFIRMED' : 'PENDING',
   492→            checkIn: checkInDate, checkOut: checkOutDate, nights,
   493→            adults: adults || 1, children: children || 0, infants: infants || 0, pets: pets || 0,
   494→            nightlySubtotal, cleaningFee, serviceFee, taxes, discounts, totalAmount, hostPayout, platformFee,
   495→            currency: 'USD', couponCode: couponCode || null,
   496→            confirmedAt: listing.instantBook ? new Date() : null,
   497→          },
   498→        })
   499→        await db.paymentAttempt.create({
   500→          data: { bookingId: booking.id, provider: 'MOCK', providerRef: `pi_test_${booking.id.slice(0, 8)}`, amount: totalAmount, currency: 'USD', status: 'CAPTURED', intent: 'CHARGE', idempotencyKey: `idem-${booking.id}` },
   501→        })
   502→        if (listing.instantBook) {
   503→          const scheduledAt = new Date(checkOutDate.getTime() + 24 * 3600 * 1000)
   504→          await db.payout.create({ data: { hostId: booking.hostId, bookingId: booking.id, amount: hostPayout, currency: 'USD', status: 'PENDING', provider: 'BANK', scheduledAt } })
   505→        }
   506→        await db.notification.create({ data: { userId: me.id, type: 'BOOKING_CONFIRMED', title: listing.instantBook ? 'Your booking is confirmed' : 'Your booking request was sent', body: `${listing.title} • ${checkInDate.toDateString()} → ${checkOutDate.toDateString()}`, link: '/guest?tab=trips' } })
   507→        await db.notification.create({ data: { userId: listing.hostId, type: 'BOOKING_REQUEST_RECEIVED', title: listing.instantBook ? 'New booking confirmed' : 'New booking request', body: `${me.name} booked ${listing.title}`, link: '/host?tab=reservations' } })
   508→        return NextResponse.json({ booking })
   509→      }
   510→      case 'cancel-booking': {
   511→        const me = await requireAuth()
   512→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   513→        const { bookingId, reason } = body
   514→        const booking = await db.booking.findUnique({ where: { id: bookingId } })
   515→        if (!booking) return NextResponse.json({ error: 'not found' }, { status: 404 })
   516→        if (booking.guestId !== me.id && booking.hostId !== me.id && me.role !== 'ADMIN') return NextResponse.json({ error: 'forbidden' }, { status: 403 })
   517→        const updated = await db.booking.update({ where: { id: bookingId }, data: { status: 'CANCELLED', cancelledAt: new Date(), cancellationReason: reason || 'Cancelled by user', cancellationPolicyApplied: 'FLEXIBLE' } })
   518→        await db.paymentAttempt.updateMany({ where: { bookingId, status: 'CAPTURED' }, data: { status: 'REFUNDED' } })
   519→        const payment = await db.paymentAttempt.findFirst({ where: { bookingId } })
   520→        if (payment) {
   521→          await db.refund.create({ data: { paymentAttemptId: payment.id, amount: booking.totalAmount, currency: booking.currency, reason: reason || 'Cancelled by user', status: 'COMPLETED' } })
   522→        }
   523→        await db.payout.deleteMany({ where: { bookingId } })
   524→        await db.notification.create({ data: { userId: booking.guestId, type: 'RESERVATION_CANCELLED', title: 'Your reservation was cancelled', body: `Refund of $${(booking.totalAmount / 100).toFixed(2)} processed`, link: '/guest?tab=trips' } })
   525→        return NextResponse.json({ booking: updated })
   526→      }
   527→      case 'add-wishlist': {
   528→        const me = await requireAuth()
   529→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   530→        const { listingId, wishlistId, name } = body
   531→        let wId = wishlistId
   532→        if (!wId) {
   533→          const w = await db.wishlist.create({ data: { userId: me.id, name: name || 'Saved' } })
   534→          wId = w.id
   535→        }
   536→        await db.wishlist.update({ where: { id: wId }, data: { listings: { connect: { id: listingId } } } })
   537→        await db.listing.update({ where: { id: listingId }, data: { wishlistSaves: { increment: 1 } } })
   538→        return NextResponse.json({ wishlistId: wId })
   539→      }
   540→      case 'remove-wishlist': {
   541→        const me = await requireAuth()
   542→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   543→        const { listingId, wishlistId } = body
   544→        await db.wishlist.update({ where: { id: wishlistId }, data: { listings: { disconnect: { id: listingId } } } })
   545→        await db.listing.update({ where: { id: listingId }, data: { wishlistSaves: { decrement: 1 } } })
   546→        return NextResponse.json({ ok: true })
   547→      }
   548→      case 'send-message': {
   549→        const me = await requireAuth()
   550→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   551→        const { conversationId, recipientId, body: msgBody, bookingId } = body
   552→        if (!msgBody) return NextResponse.json({ error: 'message body required' }, { status: 400 })
   553→        let convId = conversationId
   554→        if (!convId && recipientId) {
   555→          const existing = await db.conversation.findFirst({ where: { OR: [{ userAId: me.id, userBId: recipientId }, { userAId: recipientId, userBId: me.id }] } })
   556→          if (existing) convId = existing.id
   557→          else {
   558→            const conv = await db.conversation.create({ data: { userAId: me.id, userBId: recipientId, bookingId: bookingId || null } })
   559→            convId = conv.id
   560→          }
   561→        }
   562→        if (!convId) return NextResponse.json({ error: 'conversation required' }, { status: 400 })
   563→        const message = await db.message.create({ data: { conversationId: convId, senderId: me.id, bookingId: bookingId || null, body: msgBody, isRead: false, sentAt: new Date() } })
   564→        await db.conversation.update({ where: { id: convId }, data: { lastMessageAt: new Date() } })
   565→        const conv = await db.conversation.findUnique({ where: { id: convId } })
   566→        if (conv) {
   567→          const recipientId = conv.userAId === me.id ? conv.userBId : conv.userAId
   568→          await db.notification.create({ data: { userId: recipientId, type: 'MESSAGE_RECEIVED', title: `New message from ${me.name}`, body: msgBody.slice(0, 80), link: '/guest?tab=messages' } })
   569→        }
   570→        return NextResponse.json({ message, conversationId: convId })
   571→      }
   572→      case 'mark-read': {
   573→        const me = await requireAuth()
   574→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   575→        const { conversationId } = body
   576→        await db.message.updateMany({ where: { conversationId, senderId: { not: me.id }, isRead: false }, data: { isRead: true, readAt: new Date() } })
   577→        return NextResponse.json({ ok: true })
   578→      }
   579→      case 'create-review': {
   580→        const me = await requireAuth()
   581→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   582→        const { bookingId, rating, body: reviewBody, subratings } = body
   583→        const booking = await db.booking.findUnique({ where: { id: bookingId }, include: { listing: true } })
   584→        if (!booking) return NextResponse.json({ error: 'booking not found' }, { status: 404 })
   585→        if (booking.guestId !== me.id) return NextResponse.json({ error: 'forbidden' }, { status: 403 })
   586→        const existing = await db.review.findFirst({ where: { bookingId } })
   587→        if (existing) return NextResponse.json({ error: 'review already submitted' }, { status: 409 })
   588→        const review = await db.review.create({
   589→          data: {
   590→            authorId: me.id, subjectId: booking.hostId, listingId: booking.listingId, bookingId, rating,
   591→            ratingAccuracy: subratings?.accuracy, ratingCleanliness: subratings?.cleanliness,
   592→            ratingCommunication: subratings?.communication, ratingLocation: subratings?.location,
   593→            ratingCheckIn: subratings?.checkIn, ratingValue: subratings?.value, body: reviewBody || null,
   594→          },
   595→        })
   596→        const reviews = await db.review.findMany({ where: { listingId: booking.listingId, status: 'PUBLISHED' } })
   597→        const avgRating = reviews.reduce((s, r) => s + r.rating, 0) / reviews.length
   598→        await db.listing.update({ where: { id: booking.listingId }, data: { rating: Math.round(avgRating * 10) / 10, reviewCount: reviews.length } })
   599→        return NextResponse.json({ review })
   600→      }
   601→      case 'mark-notification-read': {
   602→        const me = await requireAuth()
   603→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   604→        const { notificationId, all } = body
   605→        if (all) await db.notification.updateMany({ where: { userId: me.id }, data: { isRead: true } })
   606→        else await db.notification.update({ where: { id: notificationId }, data: { isRead: true } })
   607→        return NextResponse.json({ ok: true })
   608→      }
   609→      case 'create-listing': {
   610→        const me = await requireAuth(['HOST', 'ADMIN'])
   611→        if (!me) return NextResponse.json({ error: 'host account required' }, { status: 403 })
   612→        const { id, ...data } = body
   613→        const slug = data.title ? `${data.title.toLowerCase().replace(/[^a-z0-9]+/g, '-')}-${Date.now().toString().slice(-4)}` : `listing-${Date.now()}`
   614→        if (id) {
   615→          const listing = await db.listing.update({ where: { id }, data })
   616→          return NextResponse.json({ listing })
   617→        }
   618→        const listing = await db.listing.create({
   619→          data: {
   620→            hostId: me.id,
   621→            title: data.title || 'Untitled Listing', slug,
   622→            description: data.description || '',
   623→            category: data.category || 'APARTMENT', roomType: data.roomType || 'ENTIRE',
   624→            status: data.status || 'DRAFT',
   625→            country: data.country || 'Portugal', city: data.city || 'Lisbon', neighborhood: data.neighborhood || null,
   626→            lat: data.lat || null, lng: data.lng || null,
   627→            maxGuests: data.maxGuests || 2, bedrooms: data.bedrooms || 1, beds: data.beds || 1, bathrooms: data.bathrooms || 1,
   628→            basePrice: data.basePrice || 5000, cleaningFee: data.cleaningFee || 0,
   629→            minNights: data.minNights || 1,
   630→            amenities: data.amenities || '',
   631→            coverImage: data.coverImage || 'https://images.unsplash.com/photo-1502672260266-1c0e6d6f3a9b?w=1200&q=80',
   632→            allowsPets: data.allowsPets || false, allowsSmoking: data.allowsSmoking || false, allowsParties: data.allowsParties || false,
   633→            cancellationPolicy: data.cancellationPolicy || 'FLEXIBLE', instantBook: data.instantBook || false,
   634→          },
   635→        })
   636→        return NextResponse.json({ listing })
   637→      }
   638→      case 'update-listing-status': {
   639→        const me = await requireAuth(['HOST', 'ADMIN'])
   640→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   641→        const { listingId, status } = body
   642→        const listing = await db.listing.findUnique({ where: { id: listingId } })
   643→        if (!listing || (listing.hostId !== me.id && me.role !== 'ADMIN')) return NextResponse.json({ error: 'forbidden' }, { status: 403 })
   644→        const updated = await db.listing.update({ where: { id: listingId }, data: { status } })
   645→        return NextResponse.json({ listing: updated })
   646→      }
   647→      case 'admin-user-action': {
   648→        const me = await requireAuth(['ADMIN'])
   649→        if (!me) return NextResponse.json({ error: 'forbidden' }, { status: 403 })
   650→        const { userId, action } = body
   651→        if (action === 'SUSPEND') await db.user.update({ where: { id: userId }, data: { isVerified: false } })
   652→        else if (action === 'REACTIVATE' || action === 'VERIFY') await db.user.update({ where: { id: userId }, data: { isVerified: true } })
   653→        else if (action === 'DELETE') await db.user.delete({ where: { id: userId } })
   654→        await db.auditLog.create({ data: { actorId: me.id, action: `USER_${action}`, resource: 'User', resourceId: userId, ip: '127.0.0.1' } })
   655→        return NextResponse.json({ ok: true })
   656→      }
   657→      case 'admin-listing-action': {
   658→        const me = await requireAuth(['ADMIN'])
   659→        if (!me) return NextResponse.json({ error: 'forbidden' }, { status: 403 })
   660→        const { listingId, action } = body
   661→        const moderationStatus = action === 'APPROVE' ? 'AUTO_APPROVED' : action === 'REJECT' ? 'REJECTED' : 'RESTRICTED'
   662→        const status = action === 'REMOVE' ? 'ARCHIVED' : action === 'APPROVE' ? 'PUBLISHED' : 'PAUSED'
   663→        const updated = await db.listing.update({ where: { id: listingId }, data: { moderationStatus, status } })
   664→        await db.auditLog.create({ data: { actorId: me.id, action: `LISTING_${action}`, resource: 'Listing', resourceId: listingId, ip: '127.0.0.1' } })
   665→        return NextResponse.json({ listing: updated })
   666→      }
   667→      case 'admin-refund': {
   668→        const me = await requireAuth(['ADMIN'])
   669→        if (!me) return NextResponse.json({ error: 'forbidden' }, { status: 403 })
   670→        const { bookingId, amount } = body
   671→        const payment = await db.paymentAttempt.findFirst({ where: { bookingId } })
   672→        if (!payment) return NextResponse.json({ error: 'no payment found' }, { status: 404 })
   673→        await db.refund.create({ data: { paymentAttemptId: payment.id, amount: amount || payment.amount, currency: payment.currency, status: 'COMPLETED', reason: 'Admin-issued refund' } })
   674→        await db.paymentAttempt.update({ where: { id: payment.id }, data: { status: 'REFUNDED' } })
   675→        await db.auditLog.create({ data: { actorId: me.id, action: 'REFUND_ISSUED', resource: 'Booking', resourceId: bookingId, ip: '127.0.0.1' } })
   676→        return NextResponse.json({ ok: true })
   677→      }
   678→      case 'support-ticket': {
   679→        const me = await requireAuth()
   680→        if (!me) return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
   681→        const { subject, category, description, priority, bookingId } = body
   682→        const ticket = await db.supportTicket.create({
   683→          data: { requesterId: me.id, subject, category: category || 'GENERAL', priority: priority || 'NORMAL', status: 'OPEN', description, bookingId: bookingId || null },
   684→        })
   685→        await db.notification.create({ data: { userId: me.id, type: 'SUPPORT_UPDATE', title: 'Support ticket created', body: `Ticket #${ticket.id.slice(0, 8)} — we'll respond within 24 hours.`, link: '/guest?tab=support' } })
   686→        return NextResponse.json({ ticket })
   687→      }
   688→      default:
   689→        return NextResponse.json({ error: 'unknown op' }, { status: 400 })
   690→    }
   691→  } catch (err: any) {
   692→    console.error('[stayora API POST error]', op, err)
   693→    return NextResponse.json({ error: err.message || 'server error' }, { status: 500 })
   694→  }
   695→}

   697→async function createSession(userId: string): Promise<string> {
   698→  const token = `${Date.now().toString(36)}-${Math.random().toString(36).slice(2)}-${userId}`
   699→  await db.session.create({ data: { userId, token, expiresAt: new Date(Date.now() + 30 * 86400000) } })
   700→  return token
   701→}
