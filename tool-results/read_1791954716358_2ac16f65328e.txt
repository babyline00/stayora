   262→   262→function TripsTab({ bookings, loading, onRefresh }: any) {

   264→   264→  const [cancelling, setCancelling] = useState<string | null>(null)
   265→   265→

   267→   267→  const past = bookings.filter((b: any) => b.status === 'COMPLETED')
   268→   268→  const cancelled = bookings.filter((b: any) => b.status === 'CANCELLED')
   269→   269→
   270→   270→  const handleCancel = async (bookingId: string) => {
   271→   271→    setCancelling(bookingId)

   274→   274→      showToast('Booking cancelled', 'Your refund is processing.', 'success')
   275→   275→      onRefresh()
   276→   276→    } catch (e: any) {
   277→   277→      showToast('Could not cancel', e.message, 'error')
   278→   278→    } finally {
   279→   279→      setCancelling(null)
   280→   280→    }
   281→   281→  }
   282→   282→
   283→   283→  if (loading) return <Skeleton className="h-96 w-full rounded-2xl" />
   284→   284→
   285→   285→  return (
   286→   286→    <div className="space-y-8">
   287→   287→      <div>
   288→   288→        <h1 className="text-2xl font-bold text-navy">My Trips</h1>
   289→   289→        <p className="text-sm text-muted-foreground">Manage your upcoming and past reservations.</p>
   290→   290→      </div>
   291→   291→
   292→   292→      <TripSection title="Upcoming" bookings={upcoming} cancelling={cancelling} onCancel={handleCancel} onGoToListing={(id: string) => { selectListing(id); setView('property') }} />
